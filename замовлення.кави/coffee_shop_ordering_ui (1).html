Приклад C++ Структур (Data Models):

#include <string>
#include <vector>

struct Customization {
    std::string size;        // "S", "M", "L"
    std::string milk_type;   // "Oat", "Almond", "Normal"
    int sugar_cubes;        // 0, 1, 2, 3
    std::vector<std::string> syrups;
};

struct OrderItem {
    int item_id;
    std::string name;
    double unit_price;
    int quantity;
    Customization options;
};

struct CoffeeOrderPayload {
    int order_id;
    std::string customer_name;
    std::string order_type;  // "takeaway" / "dinein"
    int table_number;
    std::string payment_method;
    double total_price_uah;
    std::vector<OrderItem> cart_items;
};
